machine  car-radio  "Car radio"
format   1
initial  off

states
  off        "Off"
  radio      "Radio"
  seeking    "Seeking"
  cd         "CD"
  bluetooth  "Bluetooth"

events
  power   user
  source  user
  seek    user
  cancel  user
  eject   user
  giveup  runtime  "no station found"

config
  has-cd-slot   = true  "This trim has a CD slot"
  seek-seconds  = 8     "Seek gives up after (s)"

facts
  disc-loaded       "A disc is in the slot"
  bluetooth-paired  "A phone is paired"

refusals
  no-disc     "there is no disc in the slot"
  no-phone    "no phone is paired"
  not-fitted  "this car has no CD slot"

off:
  power                                -> radio
  source, seek, cancel, eject, giveup  -> stay

radio:
  power                                                               -> off
  seek                                                                -> seeking
  source                               | has-cd-slot and disc-loaded  -> cd
                                       | not has-cd-slot              -> refuse not-fitted
                                       | not disc-loaded              -> refuse no-disc
                                       | otherwise                    -> stay
  cancel, giveup                                                      -> stay
  eject                                                               -> refuse no-disc

seeking:  after seek-seconds fires giveup
  giveup, cancel                       -> radio
  seek, source                         -> stay
  power                                -> off
  eject                                -> refuse no-disc

cd:
  power                                                    -> off
  eject                                                    -> radio
  source                               | bluetooth-paired  -> bluetooth
                                       | otherwise         -> refuse no-phone
  seek, cancel, giveup                                     -> stay

bluetooth:
  power                                -> off
  source                               -> radio
  eject                                -> refuse no-disc
  seek, cancel, giveup                 -> stay
