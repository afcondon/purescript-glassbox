format   1
machine  washing-machine
title    "Washing machine"
initial  idle

state    idle                      "Idle"
state    locking                   "Locking"
state    prewashing                "Prewash"
state    filling                   "Filling"
state    washing                   "Washing"
state    draining                  "Draining"
state    rinse-filling             "Rinse fill"
state    rinsing                   "Rinsing"
state    spinning                  "Spinning"
state    unlocking                 "Unlocking"
state    done             final    "Done"
state    paused                    "Paused"
event    start            user
event    pause            user
event    resume           user
event    cancel           user
event    open             user     "open door"
event    step-done        runtime  "step complete"
config   has-prewash      = false  "Prewash in this programme"
fact     door-shut                 "The door is shut"
fact     more-rinses               "The controller wants another rinse"
fact     drum-wet                  "There is water in the drum"
fact     paused-in-fill            "Paused while filling"
fact     paused-in-rinse           "Paused while rinsing"
refusal  door-ajar                 "the door is not shut"
refusal  door-locked               "the door is locked while the drum is running"
refusal  still-wet                 "the drum must drain before the door opens"
refusal  not-running               "nothing is running"

idle           start      when not door-shut    refuse door-ajar
idle           start      else                  -> locking
idle           open                             stay
idle           pause                            refuse not-running
idle           resume                           refuse not-running
idle           cancel                           stay
idle           step-done                        stay
locking        step-done  when has-prewash      -> prewashing
locking        step-done  else                  -> filling
locking        cancel                           -> unlocking
locking        open                             refuse door-locked
locking        pause                            -> paused
locking        resume                           stay
locking        start                            stay
prewashing     step-done                        -> draining
prewashing     pause                            -> paused
prewashing     cancel                           -> draining
prewashing     open                             refuse door-locked
prewashing     resume                           stay
prewashing     start                            stay
filling        step-done                        -> washing
filling        pause                            -> paused
filling        cancel                           -> draining
filling        open                             refuse door-locked
filling        resume                           stay
filling        start                            stay
washing        step-done                        -> draining
washing        pause                            -> paused
washing        cancel                           -> draining
washing        open                             refuse door-locked
washing        resume                           stay
washing        start                            stay
draining       step-done  when more-rinses      -> rinse-filling
draining       step-done  else                  -> spinning
draining       pause                            stay
draining       cancel                           stay
draining       open                             refuse door-locked
draining       resume                           stay
draining       start                            stay
rinse-filling  step-done                        -> rinsing
rinse-filling  pause                            -> paused
rinse-filling  cancel                           -> draining
rinse-filling  open                             refuse door-locked
rinse-filling  resume                           stay
rinse-filling  start                            stay
rinsing        step-done                        -> draining
rinsing        pause                            -> paused
rinsing        cancel                           -> draining
rinsing        open                             refuse door-locked
rinsing        resume                           stay
rinsing        start                            stay
spinning       step-done                        -> unlocking
spinning       pause                            -> paused
spinning       cancel                           -> unlocking
spinning       open                             refuse door-locked
spinning       resume                           stay
spinning       start                            stay
unlocking      step-done  when drum-wet         refuse still-wet
unlocking      step-done  else                  -> done
unlocking      open                             refuse door-locked
unlocking      pause                            stay
unlocking      resume                           stay
unlocking      cancel                           stay
unlocking      start                            stay
done           open                             -> idle
done           start                            -> locking
done           pause                            refuse not-running
done           resume                           refuse not-running
done           cancel                           stay
done           step-done                        stay
paused         resume     when paused-in-rinse  -> rinsing
paused         resume     elif paused-in-fill   -> filling
paused         resume     else                  -> washing
paused         cancel                           -> draining
paused         open                             refuse door-locked
paused         pause                            stay
paused         start                            stay
paused         step-done                        stay
