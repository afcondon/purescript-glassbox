format   1
machine  car-radio
title    "Car radio"
initial  off

state     off                        "Off"
state     radio                      "Radio"
state     seeking                    "Seeking"
state     cd                         "CD"
state     bluetooth                  "Bluetooth"
event     power             user
event     source            user
event     seek              user
event     cancel            user
event     eject             user
event     giveup            runtime  "no station found"
config    has-cd-slot       = true   "This trim has a CD slot"
config    seek-seconds      = 8      "Seek gives up after (s)"
fact      disc-loaded                "A disc is in the slot"
fact      bluetooth-paired           "A phone is paired"
refusal   no-disc                    "there is no disc in the slot"
refusal   no-phone                   "no phone is paired"
refusal   not-fitted                 "this car has no CD slot"
deadline  seeking                    after seek-seconds fires giveup

off        power                                     -> radio
off        source                                    stay
off        seek                                      stay
off        cancel                                    stay
off        eject                                     stay
off        giveup                                    stay
radio      power                                     -> off
radio      seek                                      -> seeking
radio      source  when has-cd-slot and disc-loaded  -> cd
radio      source  elif not has-cd-slot              refuse not-fitted
radio      source  elif not disc-loaded              refuse no-disc
radio      source  else                              stay
radio      cancel                                    stay
radio      eject                                     refuse no-disc
radio      giveup                                    stay
seeking    giveup                                    -> radio
seeking    cancel                                    -> radio
seeking    seek                                      stay
seeking    power                                     -> off
seeking    source                                    stay
seeking    eject                                     refuse no-disc
cd         power                                     -> off
cd         eject                                     -> radio
cd         source  when bluetooth-paired             -> bluetooth
cd         source  else                              refuse no-phone
cd         seek                                      stay
cd         cancel                                    stay
cd         giveup                                    stay
bluetooth  power                                     -> off
bluetooth  source                                    -> radio
bluetooth  eject                                     refuse no-disc
bluetooth  seek                                      stay
bluetooth  cancel                                    stay
bluetooth  giveup                                    stay
