format   1
machine  loop
title    "One loop of a six-loop looper"
initial  empty

state     empty                     "Empty"
state     armed                     "Armed"
state     recording                 "Recording"
state     playing                   "Playing"
state     overdubbing               "Overdub"
state     stopped                   "Stopped"
event     record           user
event     play             user
event     stop             user
event     clear            user
event     elapsed          runtime  "count-in"
config    count-in-beats   = 4      "Count-in (beats)"
fact      converter-held            "Another loop holds the converter"
refusal   converter-busy            "another loop holds the converter"
refusal   nothing-yet               "nothing recorded yet"
refusal   overdub-in-play           "stopped loops overdub from Playing"
deadline  armed                     after count-in-beats fires elapsed

empty        record   when converter-held      refuse converter-busy
empty        record   elif count-in-beats > 0  -> armed
empty        record   else                     -> recording
empty        play                              refuse nothing-yet
empty        stop                              stay
empty        clear                             stay
empty        elapsed                           stay
armed        elapsed                           -> recording
armed        record                            -> empty
armed        stop                              -> empty
armed        clear                             -> empty
armed        play                              stay
recording    record                            -> playing
recording    play                              -> playing
recording    stop                              -> stopped
recording    clear                             -> empty
recording    elapsed                           stay
playing      record                            -> overdubbing
playing      stop                              -> stopped
playing      clear                             -> empty
playing      play                              stay
playing      elapsed                           stay
overdubbing  record                            -> playing
overdubbing  play                              -> playing
overdubbing  stop                              -> stopped
overdubbing  clear                             -> empty
overdubbing  elapsed                           stay
stopped      play                              -> playing
stopped      clear                             -> empty
stopped      record                            refuse overdub-in-play
stopped      stop                              stay
stopped      elapsed                           stay
